% Construction : a = (1, 3), b = (0, 0), c = (5, 0).
% Les milieux m = (0,5 ; 1,5) de [ab] et n = (3 ; 1,5) de [ac].
\begin{tikzpicture}[scale=0.85]
  \coordinate (a) at (1, 3);
  \coordinate (b) at (0, 0);
  \coordinate (c) at (5, 0);
  \coordinate (m) at (0.5, 1.5);
  \coordinate (n) at (3, 1.5);
  \draw (a) -- (b) -- (c) -- cycle;
  \draw[dashed] (m) -- (n);
  \foreach \p in {a, b, c, m, n} \fill (\p) circle (1.4pt);
  \node[above] at (a) {$a$};
  \node[below left] at (b) {$b$};
  \node[below right] at (c) {$c$};
  \node[left] at (m) {$m$};
  \node[right] at (n) {$n$};
\end{tikzpicture}
